## 静态成员函数
```cpp
class Demo{
private:
    static unsigned int count;
    unsigned int a;
public:
    static void hello();
};
```
```cpp
unsigned int Demo::count = 0;
void Demo::hello(){
    count++;
    a = 2;
}
```

# this指针
```cpp
class Complex{
private:
    float r;
    float i;
public:
    Complex(float r,float i){
        this->r= r;
        (*this).i= i;
    };
    Complex& add(const Complex& other){

        i+=other.i;
        r+=other.r;
        return *this;
    }
    void print() const{
        cout<<r<<"+"<<i<<"i"<<endl;
    }
};
```
- 在子类中访问父类的函数
```cpp
class BaseClass{
public:
    virtual void hello(){
        cout<<"Hello from Base"<<endl;
    }
};

class SubClass:public BaseClass{
public:
    void hello(){
        cout<<"Hello from Subclass"<<endl;
        BaseClass::hello();
    }
};
```
## 友元
```cpp
class B;

class A{
public:
    A():m_value(0){};
private:
    int m_value;
    friend class B;
};

class B{
public:
    int inc(A& obj)
    {
        obj.m_value++;
        return obj.m_value;
    }
};
```
